cmake_minimum_required(VERSION 3.16)
project(Probe CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
if(APPLE)
  set(WrapOpenGL_AGL "-framework OpenGL" CACHE STRING "" FORCE)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(probe probe.cpp ../../../../Examples/MainWindow.cpp ../../../../Examples/TableModelFactory.cpp ../../../../Examples/TreeModelFactory.cpp ../../../../Examples/ui/MainWindow.ui)
target_include_directories(probe PRIVATE ../../../../Examples)
set_target_properties(probe PROPERTIES AUTOUIC_SEARCH_PATHS ../../../../Examples/ui)
target_link_libraries(probe Qt6::Widgets)
