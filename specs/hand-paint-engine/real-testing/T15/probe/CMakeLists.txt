cmake_minimum_required(VERSION 3.21)
project(T15Probe CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(WT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../..)
add_executable(t15_probe probe.cpp)
target_include_directories(t15_probe PRIVATE ${WT}/Examples/src ${WT}/hpq/include)
target_link_libraries(t15_probe PRIVATE ${WT}/build/debug/Examples/libHandPaintingExampleLib.a ${WT}/build/debug/hpq/libhpq.a Qt6::Widgets)
add_executable(menubar_probe menubar_probe.cpp)
target_include_directories(menubar_probe PRIVATE ${WT}/hpq/include)
target_link_libraries(menubar_probe PRIVATE ${WT}/build/debug/hpq/libhpq.a Qt6::Widgets)
