cmake_minimum_required(VERSION 3.21)
project(probe CXX)
set(HPQ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../..")
find_package(Qt6 REQUIRED COMPONENTS Gui)
add_executable(probe main.cpp)
target_compile_features(probe PRIVATE cxx_std_17)
target_include_directories(probe PRIVATE ${HPQ_ROOT}/hpq/include)
target_link_libraries(probe PRIVATE ${HPQ_ROOT}/build/debug/hpq/libhpq.a Qt6::Gui)
add_executable(defaults defaults.cpp)
target_compile_features(defaults PRIVATE cxx_std_17)
target_include_directories(defaults PRIVATE ${HPQ_ROOT}/hpq/include)
target_link_libraries(defaults PRIVATE ${HPQ_ROOT}/build/debug/hpq/libhpq.a Qt6::Gui)
