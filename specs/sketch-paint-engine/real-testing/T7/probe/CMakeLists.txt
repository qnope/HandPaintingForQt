cmake_minimum_required(VERSION 3.16)
project(T7Probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Widgets)
find_package(GTest CONFIG REQUIRED)
enable_testing()
add_subdirectory(${HPQ_SOURCE_DIR}/hpq hpq)
add_executable(probe main.cpp)
target_link_libraries(probe PRIVATE hpq::hpq Qt6::Gui Qt6::Widgets)
add_executable(identity identity.cpp)
target_link_libraries(identity PRIVATE hpq::hpq Qt6::Gui Qt6::Widgets)
