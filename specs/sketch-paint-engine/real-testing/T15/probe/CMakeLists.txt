cmake_minimum_required(VERSION 3.16)
project(T15Probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE _sdk OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT EXISTS "${_sdk}/System/Library/Frameworks/AGL.framework")
    set(WrapOpenGL_AGL "-framework OpenGL")
  endif()
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(B ${HPQ_SOURCE}/build/debug)
add_executable(T15Probe Probe.cpp ${B}/hpq/CMakeFiles/hpq_resources_1.dir/.qt/rcc/qrc_hpq_fonts_init.cpp.o)
target_include_directories(T15Probe PRIVATE ${HPQ_SOURCE}/Examples/src ${HPQ_SOURCE}/hpq/include)
target_link_libraries(T15Probe PRIVATE ${B}/Examples/libHandPaintingExampleLib.a ${B}/hpq/libhpq.a Qt6::Widgets)
