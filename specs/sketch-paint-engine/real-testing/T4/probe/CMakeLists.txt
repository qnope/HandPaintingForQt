cmake_minimum_required(VERSION 3.16)
project(T4Probe LANGUAGES CXX)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE _sdk OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT EXISTS "${_sdk}/System/Library/Frameworks/AGL.framework")
    set(WrapOpenGL_AGL "-framework OpenGL")
  endif()
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(HPQ_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../..)
add_executable(probe main.cpp)
target_compile_features(probe PRIVATE cxx_std_17)
target_include_directories(probe PRIVATE ${HPQ_ROOT}/hpq/include)
target_link_libraries(probe PRIVATE ${HPQ_ROOT}/build/debug/hpq/libhpq.a Qt6::Widgets)
